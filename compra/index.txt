(function () {
  const items = document.querySelectorAll('.reveal');
  if (!('IntersectionObserver' in window)) {
    items.forEach(el => el.classList.add('is-visible'));
    return;
  }

  const io = new IntersectionObserver((entries) => {
    entries.forEach(entry => {
      if (entry.isIntersecting) {
        entry.target.classList.add('is-visible');
        io.unobserve(entry.target);
      }
    });
  }, { threshold: 0.12, rootMargin: '0px 0px -60px 0px' });

  items.forEach(el => io.observe(el));
})();


/* ---------------------------------------------------------
   C. CONTADORES ANIMADOS
   --------------------------------------------------------- */
(function () {
  const counters = document.querySelectorAll('[data-counter]');
  if (!counters.length) return;

  const easeOutCubic = t => 1 - Math.pow(1 - t, 3);

  const runCounter = (el) => {
    const target   = parseFloat(el.dataset.counter) || 0;
    const suffix   = el.dataset.suffix || '';
    const prefix   = el.dataset.prefix || '';
    const duration = 1800;
    const start    = performance.now();

    const step = (now) => {
      const progress = Math.min((now - start) / duration, 1);
      const value    = target * easeOutCubic(progress);

      el.textContent = prefix +
        Math.round(value).toLocaleString('es-DO') +
        suffix;

      if (progress < 1) requestAnimationFrame(step);
      else el.textContent = prefix + target.toLocaleString('es-DO') + suffix;
    };

    requestAnimationFrame(step);
  };

  if (!('IntersectionObserver' in window)) {
    counters.forEach(runCounter);
    return;
  }

  const io = new IntersectionObserver((entries) => {
    entries.forEach(entry => {
      if (entry.isIntersecting) {
        runCounter(entry.target);
        io.unobserve(entry.target);
      }
    });
  }, { threshold: 0.5 });

  counters.forEach(el => io.observe(el));
})();


/* ---------------------------------------------------------
   D. TILT CARDS + GLOW QUE SIGUE AL CURSOR
   --------------------------------------------------------- */
(function () {
  const canHover = window.matchMedia('(hover: hover) and (pointer: fine)').matches;
  if (!canHover) return;

  const MAX_TILT = 7; // grados

  document.querySelectorAll('.tilt-card').forEach(card => {
    let raf = null;

    const onMove = (e) => {
      if (raf) return;

      raf = requestAnimationFrame(() => {
        const rect = card.getBoundingClientRect();
        const x = e.clientX - rect.left;
        const y = e.clientY - rect.top;

        const px = x / rect.width;   // 0 → 1
        const py = y / rect.height;  // 0 → 1

        const rotateY = (px - 0.5) * MAX_TILT * 2;
        const rotateX = (0.5 - py) * MAX_TILT * 2;

        card.style.setProperty('--mx', x + 'px');
        card.style.setProperty('--my', y + 'px');

        card.style.transform =
          `perspective(1000px) rotateX(${rotateX.toFixed(2)}deg) ` +
          `rotateY(${rotateY.toFixed(2)}deg) translateY(-6px)`;

        raf = null;
      });
    };

    const onLeave = () => {
      if (raf) { cancelAnimationFrame(raf); raf = null; }
      card.style.transform = '';
      card.style.removeProperty('--mx');
      card.style.removeProperty('--my');
    };

    card.addEventListener('mousemove', onMove);
    card.addEventListener('mouseleave', onLeave);
  });
})();

      if (TEST_MODE) return;
      try {
        sessionStorage.setItem(STORAGE_KEY, '1');
      } catch (e) {}
    }

    function openExitModal() {
      if (modalOverlay.classList.contains('is-open')) return;
      modalOverlay.classList.add('is-open');
      document.body.style.overflow = 'hidden';
      markModalShown();
    }

    function closeExitModal() {
      modalOverlay.classList.remove('is-open');
      document.body.style.overflow = '';
    }

    modalCloseBtn.addEventListener('click', closeExitModal);

    modalOverlay.addEventListener('click', (e) => {
      if (e.target === modalOverlay) closeExitModal();
    });

    document.addEventListener('keydown', (e) => {
      if (e.key === 'Escape' && modalOverlay.classList.contains('is-open')) {
        closeExitModal();
      }
    });

    let exitIntentArmed = false;
    setTimeout(() => {
      exitIntentArmed = true;
    }, 1000);

    function handleExit() {
      if (!exitIntentArmed || hasShownModal()) return;
      if (modalOverlay.classList.contains('is-open')) return;
      openExitModal();
    }

    document.documentElement.addEventListener('mouseleave', handleExit);
    document.addEventListener('mouseout', (e) => {
      if (!e.relatedTarget && !e.toElement) {
        handleExit();
      }
    });
  })();

  /* =========================================================
     3. INTERACTIVE 3D FANNED CARD STACK
     ========================================================= */
  (function () {
    const stage = document.getElementById('cardStage');
    const stack = document.getElementById('cardStack');
    const leftCard = document.querySelector('[data-card="left"]');
    const centerCard = document.querySelector('[data-card="center"]');
    const rightCard = document.querySelector('[data-card="right"]');

    if (!stage || !stack || !leftCard || !centerCard || !rightCard) return;

    const allCards = [leftCard, centerCard, rightCard];
    const target = { x: 0, y: 0, active: 0 };
    const current = { x: 0, y: 0, active: 0 };
    const lerp = (start, end, factor) => start + (end - start) * factor;

    stage.addEventListener('mousemove', (e) => {
      const rect = stage.getBoundingClientRect();
      const nx = ((e.clientX - rect.left) / rect.width) * 2 - 1;
      const ny = ((e.clientY - rect.top) / rect.height) * 2 - 1;

      target.x = Math.max(-1, Math.min(1, nx));
      target.y = Math.max(-1, Math.min(1, ny));
      target.active = 1;

      allCards.forEach((card) => {
        const cardRect = card.getBoundingClientRect();
        const gx = ((e.clientX - cardRect.left) / cardRect.width) * 100;
        const gy = ((e.clientY - cardRect.top) / cardRect.height) * 100;
        card.style.setProperty('--glare-x', `${gx}%`);
        card.style.setProperty('--glare-y', `${gy}%`);
      });
    });

    stage.addEventListener('mouseleave', () => {
      target.x = 0;
      target.y = 0;
      target.active = 0;
    });

    function animateCards() {
      current.x = lerp(current.x, target.x, 0.08);
      current.y = lerp(current.y, target.y, 0.08);
      current.active = lerp(current.active, target.active, 0.07);

      const stackRotateY = current.x * 11;
      const stackRotateX = -current.y * 9;
      stack.style.transform = `rotateX(${stackRotateX}deg) rotateY(${stackRotateY}deg)`;

      const spread = current.active * 14;

      const leftX = -58 - spread + current.x * -10;
      const leftY = 12 + current.y * -6;
      const leftRot = -15 - current.active * 3 + current.x * 2.5;
      leftCard.style.transform = `translate3d(${leftX}px, ${leftY}px, -36px) rotate(${leftRot}deg)`;

      const rightX = 58 + spread + current.x * -10;
      const rightY = 12 + current.y * -6;
      const rightRot = 15 + current.active * 3 + current.x * 2.5;
      rightCard.style.transform = `translate3d(${rightX}px, ${rightY}px, -36px) rotate(${rightRot}deg)`;

      const centerX = current.x * 8;
      const centerY = current.y * 6;
      const centerZ = 18 + current.active * 10;
      centerCard.style.transform = `translate3d(${centerX}px, ${centerY}px, ${centerZ}px)`;

      requestAnimationFrame(animateCards);
    }

    animateCards();
  })();
</script>

        function selectPbOption(optionType) {
            const options = document.querySelectorAll('.pb-option');
            options.forEach(el => el.classList.remove('active'));

            if(selectedOption) selectedOption.classList.add('active');

                ctaButton.href = pbPaymentLinks[optionType];

        // Sincronizar la opción activa inicial (Tapa Suave por defecto) y el enlace del botón
        function syncInitialActiveOption() {
            const activeOption = document.querySelector('.pb-option.active');
            if (activeOption) {
                let optionId = activeOption.id;
                if (optionId === 'pb-opt-tapa-dura') {
                    const cta = document.getElementById('pb-cta-button');
                    if (cta && pbPaymentLinks['tapa-dura']) cta.href = pbPaymentLinks['tapa-dura'];
                } else if (optionId === 'pb-opt-tapa-suave') {
                    const cta = document.getElementById('pb-cta-button');
                    if (cta && pbPaymentLinks['tapa-suave']) cta.href = pbPaymentLinks['tapa-suave'];
                }
            } else {
                // Fallback: activar tapa suave si no hay ningún active
                const suaveOption = document.getElementById('pb-opt-tapa-suave');
                if (suaveOption) {
                    suaveOption.classList.add('active');
                    const cta = document.getElementById('pb-cta-button');
                    if (cta && pbPaymentLinks['tapa-suave']) cta.href = pbPaymentLinks['tapa-suave'];
                }
            }
        }
        
        if (document.readyState === 'loading') {
            document.addEventListener('DOMContentLoaded', syncInitialActiveOption);
        } else {
            syncInitialActiveOption();
        }
